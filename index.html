toCSV = (rows, cols) => {
  const esc = v => {
    const s = (v === null || v === undefined) ? "" : String(v)
    return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s
  }
  return [cols.join(","), ...rows.map(r => cols.map(c => esc(r[c])).join(","))]
    .join("\n")
}
// 1D Gaussian KDE. Plot.density is 2D contours and is not what we want
// here — height must encode density along a single measure.
kde = (values, bw = 0.045, n = 160) => {
  const grid = Array.from({length: n}, (_, i) => i / (n - 1))
  const c = 1 / (values.length * bw * Math.sqrt(2 * Math.PI))
  return grid.map(x => ({
    x,
    y: c * values.reduce((s, v) => {
      const z = (x - v) / bw
      return s + Math.exp(-0.5 * z * z)
    }, 0)
  }))
}

// Merge consecutive years in the same type. Runs break on a type change
// OR a gap in the series, so a missing year never reads as continuity.
runs = {
  const out = []
  let cur = null
  for (const d of trendData) {
    if (cur && d.type === cur.type && d.year === cur.y1 + 1) {
      cur.y1 = d.year; cur.n += 1
    } else {
      if (cur) out.push(cur)
      cur = {type: d.type, y0: d.year, y1: d.year, n: 1}
    }
  }
  if (cur) out.push(cur)
  return out
}

densityPanel = (m) => {
  const vals = focusRows.map(d => d[m.key])

  // Each curve is scaled to its own peak. A tight type has a density an
  // order of magnitude taller than a broad one; on a shared scale the
  // broad type flattens into the axis. Height therefore reads as relative
  // concentration within a type, not as a count.
  const norm = curve => {
    const mx = Math.max(...curve.map(p => p.y)) || 1
    return curve.map(p => ({x: p.x, y: p.y / mx}))
  }

  const focal = norm(kde(vals))

  const otherCurves = compareTypes.map(t => ({
    type: t,
    curve: norm(kde(panel.filter(d => d.type === t).map(d => d[m.key])))

        ticks: 10, tickFormat: ".1f"},

    // under the baseline rather than on the curve, so the y-scale stays
    // quantitative throughout.

layerCountries = {
  const isHybrid = layerType === hybridType
  const byCountry = new Map()
  for (const d of layerRows) {
    if (!byCountry.has(d.country_name)) byCountry.set(d.country_name, [])
    byCountry.get(d.country_name).push(d)
  }

  const rows = [...byCountry.entries()]
    .map(([name, ds]) => {
      const mode = f => {
        const m = new Map()
        for (const d of ds) m.set(d[f], (m.get(d[f]) ?? 0) + 1)
        return [...m.entries()].sort((a, b) => b[1] - a[1])[0][0] ?? ""
      }
      return {
        country: name,
        years: `${Math.min(...ds.map(d => d.year))}–${Math.max(...ds.map(d => d.year))}`,
        n: ds.length,
        direction: mode("direction"),
        ...(isHybrid ? {level: mode("level")} : {}),
        constraint: mode("constraint"),
        pct_missing_indicator: Math.round(
          100 * ds.filter(d => d.coverage === "partial").length / ds.length)
      }
    })
    .sort((a, b) => a.country.localeCompare(b.country))

  const cols = Object.keys(rows[0])
  const heads = {country: "Country", years: "Years", n: "n",
    direction: "Direction", level: "Level", constraint: "Constraint",
    pct_missing_indicator: "% missing an indicator"}

  const url = URL.createObjectURL(
    new Blob([toCSV(rows, cols)], {type: "text/csv;charset=utf-8"}))

  return html`<div style="padding:0.4rem 0.8rem;height:100%;
    display:flex;flex-direction:column;min-height:0">
    <div style="display:flex;justify-content:space-between;align-items:baseline;
      gap:1rem;font-size:0.8rem;color:#666;margin-bottom:0.5rem;flex:0 0 auto">
      <span>${rows.length} countries ·
        ${layerRows.length.toLocaleString()} country-years · most common
        value per country</span>
      <a href=${url} download=${`ned_${layerType}_countries.csv`}
        style="white-space:nowrap;font-size:0.8rem">Download CSV</a>
    </div>

    <div style="flex:1 1 auto;min-height:0;overflow-y:auto">
      <table style="border-collapse:collapse;width:100%;font-size:0.81rem">
        <thead><tr style="border-bottom:1.5px solid #ccc;text-align:left">
          ${cols.map((c, i) => html`<th style="padding:4px 8px 4px 0;
            color:#666;font-weight:600;position:sticky;top:0;background:#fff;
            text-align:${i >= 2 ? "right" : "left"};white-space:nowrap">
            ${heads[c]}</th>`)}
        </tr></thead>
        <tbody>${rows.map(r => html`<tr style="border-bottom:1px solid #f4f4f6">
          ${cols.map((c, i) => html`<td style="padding:4px 8px 4px 0;
            white-space:nowrap;text-align:${i >= 2 ? "right" : "left"};
            color:${i === 0 ? "#111" : "#555"};
            font-variant-numeric:tabular-nums">
            ${c === "pct_missing_indicator" ? r[c] + "%" : (r[c] || "—")}</td>`)}
        </tr>`)}</tbody>
      </table>
    </div>
    <div style="font-size:0.77rem;color:#888;margin-top:0.7rem;line-height:1.5;
      flex:0 0 auto">
      Direction, level and constraint vary year to year; the value shown is
      each country's most common. <b>% missing an indicator</b> is the share of
      that country's years here where at least one of the three authority
      indicators was not reported.
    </div>
  </div>`
}
